import { defineModel, fields } from '@jsango/orm';

// 1. Strongly-typed declarative schema with automatic relations
export const User = defineModel({
  name: 'User',
  table: 'users',
  fields: {
    id: fields.uuid({ primaryKey: true }),
    email: fields.string({ unique: true, maxLength: 255 }),
    fullName: fields.string(),
    role: fields.enum(['admin', 'editor', 'viewer']),
    isActive: fields.boolean({ default: true }),
  },
  relations: (rel) => ({
    orders: rel.hasMany('Order', { foreignKey: 'userId' }),
    profile: rel.hasOne('UserProfile', { foreignKey: 'userId' }),
  }),
});

// 2. High-performance AST query with batch eager loading (no N+1!)
const usersWithOrders = await User.query()
  .where('isActive', '=', true)
  .with('orders', (q) => q.where('status', '=', 'completed'))
  .orderBy('createdAt', 'desc')
  .limit(20);
import { Application } from '@jsango/middleware';
import { schema } from '@jsango/validation';

const app = new Application();

// Ultra-fast SegRadix Trie routing (>7.7M lookups/sec) with typed parameters
app.get('/api/v1/users/:id:uuid', async (ctx) => {
  const userId = ctx.request.params['id'];
  const user = await User.query().findById(userId);

  if (!user) {
    return ctx.response.notFound({ error: 'User not found' });
  }

  return ctx.response.json(user.toJSON());
});

// Scoped middleware pipeline with automatic OpenAPI schema registration
app.group('/api/v1/protected', (router) => {
  router.use(authMiddleware);
  router.post('/checkout', checkoutHandler);
});
import { createAdminResource } from '@jsango/admin-core';
import { AdminServer } from '@jsango/admin-server';
import { User } from './models/user.js';

// 1. Zero-boilerplate Admin Resource generated directly from ORM metadata
export const UserAdmin = createAdminResource(User, {
  label: 'Users & Customers',
  icon: 'users',
  searchFields: ['email', 'fullName'],
  listFields: ['id', 'email', 'role', 'isActive'],
  filters: [{ field: 'role', type: 'select' }, { field: 'isActive', type: 'boolean' }],
  export: { formats: ['csv', 'json'] },
});

// 2. Mounts interactive React SPA with CRUD, 2FA, Session Control & Audit Logs
const adminServer = new AdminServer({ resources: [UserAdmin], path: '/admin' });
app.mount(adminServer);
import { TotpService, ScryptPasswordHasher, SessionManager } from '@jsango/auth';

// RFC 6238 Time-based One-Time Password (TOTP) 2FA
const totp = new TotpService();

// Setup Two-Factor Authenticator
const secret = totp.generateSecret();
const qrUri = totp.generateKeyUri({
  secret,
  accountName: 'admin@jsango.dev',
  issuer: 'JSango Security'
});

// Validate 6-digit TOTP code during login
const isValid = totp.verifyToken({ secret, token: '482910', window: 1 });
if (!isValid) {
  throw new AuthenticationError('Invalid 2FA Verification Code');
}

import { WebSocketServer } from '@jsango/websocket';

const ws = new WebSocketServer({ path: '/realtime' });

ws.onConnection((socket, identity) => {
  socket.joinRoom(\`org:\${identity.organizationId}\`);

  socket.on('chat:send', async (payload) => {
    // Broadcast to multi-room channel with zero memory leak backpressure
    ws.broadcast(\`org:\${identity.organizationId}\`, 'chat:message', {
      from: identity.id,
      text: payload.text,
      timestamp: Date.now()
    });
  });
});

Modular Architecture

24 Cohesive Monorepo Packages

Use the full stack together or pick individual packages. Each package adheres to strict zero-circular-dependency rules.

@jsango/core Core

Application lifecycle coordinator, structured error hierarchy, and logging abstractions.

@jsango/container Core

High-performance DI container supporting transient, singleton, and scoped lifetimes.

@jsango/router Core

Segment Radix Trie router (>7.7M ops/sec) with typed constraint matching.

@jsango/http Core

Runtime-independent HTTP request/response abstractions and streaming body parsers.

@jsango/orm Data

Declarative models, AST query builder, and pure batch eager loading (.with()).

@jsango/migrations Data

Schema diffing engine, DDL compilers, distributed locks, and migration runners.

@jsango/database Data

Multi-connection manager, FIFO connection pool, and scoped transaction state machines.

@jsango/admin-ui Admin

Interactive React SPA admin dashboard with CRUD views, filters, and theme switcher.

@jsango/admin-server Admin

Admin REST orchestrator for CRUD endpoints, 2FA auth, session revoke, and export.

@jsango/admin-auth Admin

Staff authorization, resource-level CRUD permissions, and field-level visibility rules.

@jsango/auth Services

RFC 6238 TOTP 2FA engine, Session/JWT auth, Scrypt hashing, and policy RBAC.

@jsango/queue Services

At-least-once background job queues, concurrent worker polling, and dead-letter store.

@jsango/websocket Services

Multi-room WebSocket management with heartbeat monitoring and backpressure safeguards.

@jsango/openapi Services

Deterministic OpenAPI 3.1 document generation from router, validation, ORM, and Admin.

@jsango/observability Services

Structured JSON logging, Prometheus metrics, monotonic tracing, and health checks.

@jsango/cli Core

CLI developer tooling, scaffolding commands, migration generators, and dev servers.
