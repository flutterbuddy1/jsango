import { model, fields } from 'jsango';

// 1. Strongly-typed declarative model
export const User = model('User', {
  id: fields.id(),
  email: fields.string({ unique: true }),
  name: fields.string(),
  role: fields.string({ defaultValue: 'viewer' }),
  active: fields.boolean({ defaultValue: true }),
});

// 2. Query seamlessly with static model helpers
const users = await User.where('active', true)
  .orderBy('createdAt', 'desc')
  .limit(20)
  .get();

const newUser = await User.create({ name: 'Alex', email: 'alex@jsango.dev' });
import { agent, tool, object, string, number, createApp } from 'jsango';

// 1. Define type-safe tools with automatic JSON Schema generation
const lookupOrder = tool({
  name: 'lookupOrder',
  description: 'Get order status by order ID',
  schema: object({ orderId: string() }),
  execute: async ({ orderId }) => await Order.find(orderId),
});

const refundOrder = tool({
  name: 'refundOrder',
  description: 'Issue customer refund',
  schema: object({ orderId: string(), amount: number() }),
  requiresApproval: true, // Human approval gate
  execute: async ({ orderId, amount }) => await PaymentService.refund(orderId, amount),
});

// 2. Create the Autonomous Agent
export const supportAgent = agent({
  name: 'SupportAgent',
  model: 'openai:gpt-4o',
  instructions: 'Help customers check orders and handle refund requests.',
  tools: { lookupOrder, refundOrder },
});

// 3. Expose over HTTP / SSE Streaming and WebSockets in 1 line
const app = createApp();
app.agent('/api/support', supportAgent);
app.wsAgent('/ws/support', supportAgent);

await app.listen(3000);
import { createApp, validate, schema, string, email, notFound } from 'jsango';
import { User } from './models/user.js';

const app = createApp();

// Ultra-fast routing with automatic JSON responses & typed context
app.get('/users/:id', async ({ params }) => {
  const user = await User.find(params.id);
  if (!user) throw notFound('User not found');
  return user;
});

// Fluent request validation
app.post('/users', validate({
  body: schema({ name: string().min(2), email: email() })
}), async ({ body }) => {
  return User.create(body);
});

await app.listen(3000);

import { createApp } from 'jsango';

const app = createApp();

// Concise WebSocket endpoint with automatic JSON serialization & rooms
app.ws('/chat', (socket) => {
  socket.on('join', (room) => {
    socket.join(room);
    socket.to(room).send({ type: 'joined', user: socket.id });
  });

  socket.on('message', ({ room, text }) => {
    socket.to(room).send({ type: 'message', from: socket.id, text });
  });
});

await app.listen(3000);

import { TotpService } from '@jsango/auth';

// RFC 6238 Time-based One-Time Password (TOTP) 2FA
const totp = new TotpService();

// Setup Two-Factor Authenticator
const secret = totp.generateSecret();
const qrUri = totp.generateKeyUri({
  secret,
  accountName: 'admin@jsango.dev',
  issuer: 'JSango Security'
});

// Validate 6-digit TOTP code during login
const isValid = totp.verifyToken({ secret, token: '482910', window: 1 });
if (!isValid) {
  throw new AuthenticationError('Invalid 2FA Verification Code');
}

  features: { http2: true, cluster: true }
